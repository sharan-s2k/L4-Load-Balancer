cmake_minimum_required(VERSION 3.16)
project(advanced_tcp_load_balancer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(lb_core src/event_loop.cpp src/load_balancer.cpp)
target_include_directories(lb_core PUBLIC src)
target_link_libraries(lb_core PUBLIC Threads::Threads)
if(APPLE)
  target_compile_definitions(lb_core PUBLIC LB_USE_KQUEUE=1)
elseif(UNIX)
  target_compile_definitions(lb_core PUBLIC LB_USE_EPOLL=1)
else()
  message(FATAL_ERROR "Linux or macOS required")
endif()
target_compile_options(lb_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tcp-load-balancer apps/load_balancer_main.cpp)
target_link_libraries(tcp-load-balancer PRIVATE lb_core)
add_executable(echo-backend apps/echo_backend_main.cpp)
target_link_libraries(echo-backend PRIVATE Threads::Threads)
enable_testing()
add_executable(unit-tests tests/unit_tests.cpp)
target_link_libraries(unit-tests PRIVATE lb_core)
add_test(NAME unit-tests COMMAND unit-tests)
